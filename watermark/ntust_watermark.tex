% Macro command to place the watermark in the central header
\newcommand{\PlaceWaterMark}{\fancyhead[C]{\setlength{\unitlength}{1in}%
\begin{picture}(0,0)%
%\put(-2.2,-6){\usebox{\mywatermark}}% old code
\put(-0.5,-5.3){\usebox{\mywatermark}}% new code

}

\fancyhead{}  % reset left, central, right header to empty
%% If a watermark is NOT needed on every page,
%% comment out the code between >>> and <<<
%% >>> Full-document watermark
\PlaceWaterMark  % place watermark on every page (except plain and empty pages)

% Redefine the plain page style
% Also add the watermark to every plain page (first page of each chapter)

\fancypagestyle{plain}{%
\fancyhead{}%
\PlaceWaterMark%
\fancyfoot{}%
\fancyfoot[C]{\thepage}
\renewcommand{\headrulewidth}{0pt}%
\renewcommand{\footrulewidth}{0pt}%
}
%% <<< Full-document watermark

%% If only one or two pages need a watermark,
%% use \thispagestyle{WaterMarkPage} on that page (which has a page number).
%% This command does not affect the existing header or footer.
\fancypagestyle{WaterMarkPage}{%
\PlaceWaterMark%
}

%% If only one or two plain pages need a watermark (e.g. abstract, vita),
%% use \thispagestyle{PlainWaterMarkPage} on that page (which has a page number).
%% Shows only the page number and watermark, no other header or footer.
%% Equivalent to plain page style + watermark.
\fancypagestyle{PlainWaterMarkPage}{%
\fancyhead{}%
\PlaceWaterMark%
\fancyfoot{}%
\fancyfoot[C]{\thepage}
\renewcommand{\headrulewidth}{0pt}%
\renewcommand{\footrulewidth}{0pt}%
}

%% If only one or two empty pages need a watermark (e.g. cover, title page),
%% use \thispagestyle{EmptyWaterMarkPage} on that page (no page number).
%% Equivalent to empty page style + watermark.
\fancypagestyle{EmptyWaterMarkPage}{%
\fancyhead{}%
\PlaceWaterMark%
\fancyfoot{}%
\renewcommand{\headrulewidth}{0pt}%
\renewcommand{\footrulewidth}{0pt}%
}
