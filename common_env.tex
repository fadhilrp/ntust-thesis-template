%
% Reduce the top headroom above chapter titles. \@makeschapterhead covers starred
% front-matter pages (acknowledgements, table of contents, lists, abstracts) and
% \@makechapterhead covers numbered body chapters. Both bodies are copied verbatim from
% ntust_report.cls with only a negative top skip prepended. Tune the single skip value.
\makeatletter
\def\@makeschapterhead#1{%
  \vspace*{-1.5cm}%
  {\parindent \z@ \centering \normalfont
    \interlinepenalty\@M
    \LARGE \bfseries #1\par\nobreak
    \vskip 20\p@
  }}
\def\@makechapterhead#1{%
  \vspace*{-1.5cm}%
  {\parindent \z@ \centering \normalfont
    \ifnum \c@secnumdepth >\m@ne
        \LARGE\bfseries \MakeUppercase{\prechaptername}\space\thechapter
        \par\nobreak
    \fi
    \interlinepenalty\@M
    \LARGE \bfseries \MakeUppercase{#1}\par\nobreak
    \vskip 20\p@
  }}
\makeatother

% Force every chapter-level table-of-contents entry (front-matter pages and
% numbered chapters) into full caps, without touching section-level entries.
% Wrapper defined at top level; only \let happens in the begin-document hook,
% so it wraps hyperref's final \contentsline safely.
\newcommand{\ntustcapchaptertoc}[4]{%
  \ifstrequal{#1}{chapter}%
    {\ntustorigcontentsline{#1}{\MakeUppercase{#2}}{#3}{#4}}%
    {\ntustorigcontentsline{#1}{#2}{#3}{#4}}%
}
\AtBeginDocument{%
  \let\ntustorigcontentsline\contentsline
  \let\contentsline\ntustcapchaptertoc
}
